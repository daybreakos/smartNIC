# 1. Prepare performance settings (Governors, C-States, Hugepages)
sudo ./t2_perf_apply.sh

# 2. Setup Host vDPA devices
sudo ./t2_host_setup.sh --mode vdpa

# 3. Build images (if needed) & Launch VMs
./t2_build_guest_image.sh 1
./t2_build_guest_image.sh 2
./t2_launch_vm.sh 1
./t2_launch_vm.sh 2

# 4. Execute Benchmark Suite
./t2_run_benchmarks.sh

# 5. Clean up
sudo ./t2_host_reset.sh
